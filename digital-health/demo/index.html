def sd(xs):                       # 표본 표준편차 (n-1)
    m = mean(xs)
    return math.sqrt(sum((x - m) ** 2 for x in xs) / (len(xs) - 1))

def bootstrap_mean_diff_ci(a, b, B=10_000, seed=20261032, conf=0.95):
    rng = random.Random(seed)     # 시드 고정 → 누가 돌려도 같은 구간
    diffs = []
    for _ in range(B):
        sa = sum(a[int(rng.random()*len(a))] for _ in a) / len(a)
        sb = sum(b[int(rng.random()*len(b))] for _ in b) / len(b)
        diffs.append(sa - sb)
    diffs.sort()
    return quantile(diffs, 0.025), quantile(diffs, 0.975)
